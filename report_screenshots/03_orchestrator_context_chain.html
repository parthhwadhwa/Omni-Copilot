def _execute(self, plan: dict, user_input: str, log: ExecutionLog) -> dict:
    steps, results = plan.get("steps", []), []

    # Context chain — accumulates each step result to pass downstream.
    # Seeded with persistent memory (contacts, last meeting link, etc.)
    context = {**self.state.memory, **self.state.context}

    for step in steps:
        agent_name, action, params = self._normalise_step(step)
        log.add("step", f"{agent_name}.{action}")

        agent = self.agents.get(agent_name)
        if not agent:
            results.append({"ok": False, "error": f"Unknown agent: {agent_name}"})
            continue

        # Fill missing params from upstream results
        # e.g. meeting link auto-injected into the email body
        missing = _validate_and_fill(
            params, agent_name, action, context,
            last_meeting=self._last_stored("meetings.json")
        )
        if missing:
            results.append({"ok": False, "error": f"Please provide: {missing}"})
            continue

        result = agent.handle(action, params)
        results.append({**result, "status": "success" if result["ok"] else "failed"})

        # Merge result into shared context for all remaining steps
        if result.get("ok") and isinstance(result.get("result"), dict):
            context.update(result["result"])

    return {"plan": display_plan, "results": results, "log": log.entries}
