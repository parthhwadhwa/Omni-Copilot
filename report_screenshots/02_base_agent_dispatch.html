class BaseAgent:
    """Base class every agent inherits from."""
    name: str         = "base"
    description: str  = ""
    tools: list[dict] = []

    def handle(self, action: str, params: dict) -> dict:
        """
        Deterministic dispatch:
        Maps action name -> action_{action}() method on the subclass.
        Returns standardised {ok, result} or {ok, error} — never raises.
        """
        method = getattr(self, f"action_{action}", None)
        if method is None:
            return {
                "ok":    False,
                "error": f"Unknown action '{action}' for {self.name}"
            }
        try:
            return {"ok": True, "result": method(**params)}
        except Exception as e:
            return {"ok": False, "error": f"{type(e).__name__}: {e}"}

                "description":  t["description"],
                "input_schema": t["params"],
            }
            for t in self.tools
